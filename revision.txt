1. Print Syntax
print("message")
print(variable)

2. Variable Syntax
variable_name = value

a = 10
name = "Python"

3. Input Syntax
variable = input("message")
variable = int(input("message"))

4. If Syntax
if condition:
    statement

5. If Else Syntax
if condition:
    statement
else:
    statement

6. If Elif Else Syntax
if condition:
    statement
elif condition:
    statement
else:
    statement

7. For Loop Syntax
for variable in range(start, stop, step):
    statement

Example:

for i in range(1, 6):
    print(i)

8. While Loop Syntax
while condition:
    statement

9. Break Syntax
for i in range(5):
    if condition:
        break

10. Continue Syntax
for i in range(5):
    if condition:
        continue

11. Pass Syntax
for i in range(5):
    pass

12. Function Syntax
def function_name():
    statement

13. Function with Argument Syntax
def function_name(parameter):
    statement

14. Function with Return Syntax
def function_name():
    return value

15. Lambda Syntax
lambda parameter : expression

Example:

square = lambda x: x*x

16. List Syntax
list_name = [value1, value2, value3]

Access:

list_name[index]

17. Tuple Syntax
tuple_name = (value1, value2)

18. Set Syntax
set_name = {value1, value2}

19. Dictionary Syntax
dict_name = {
    key: value
}

Access:

dict_name[key]

20. String Syntax
string = "text"

Access:

string[index]

21. List Comprehension Syntax
list_name = [expression for variable in iterable]

22. Map Syntax
map(function, iterable)

Example:

list(map(function, list))

23. Filter Syntax
filter(function, iterable)

24. Recursion Syntax
def function(n):
    if condition:
        return value
    else:
        return function(n-1)

25. Assert Syntax
assert condition

26. Length Syntax
len(variable)

27. Type Syntax
type(variable)
